import org.objectweb.asm.*;
import org.objectweb.asm.tree.*;
import java.io.*;
import java.nio.file.*;
import java.util.*;
import java.util.zip.*;

public class Chk {
  static ZipFile mc; static List<Path> roots = new ArrayList<>(); static List<ZipFile> libs = new ArrayList<>();
  static Map<String,ClassNode> cache = new HashMap<>();
  static ClassNode load(String n) throws Exception {
    if (cache.containsKey(n)) return cache.get(n);
    InputStream in = null;
    ZipEntry e = mc.getEntry(n + ".class");
    if (e != null) in = mc.getInputStream(e);
    if (in == null) for (ZipFile z : libs) { e = z.getEntry(n + ".class"); if (e != null) { in = z.getInputStream(e); break; } }
    if (in == null) in = ClassLoader.getSystemResourceAsStream(n + ".class");
    if (in == null) in = Object.class.getResourceAsStream("/" + n + ".class");
    ClassNode cn = null;
    if (in != null) { cn = new ClassNode(); new ClassReader(in).accept(cn, ClassReader.SKIP_FRAMES); }
    cache.put(n, cn); return cn;
  }
  static boolean hasMethod(String cls, String name, String desc, Set<String> seen) throws Exception {
    if (cls == null || !seen.add(cls)) return false;
    ClassNode c = load(cls); if (c == null) return false;
    for (MethodNode m : c.methods) if (m.name.equals(name) && (desc == null || m.desc.equals(desc))) return true;
    if (hasMethod(c.superName, name, desc, seen)) return true;
    for (String i : c.interfaces) if (hasMethod(i, name, desc, seen)) return true;
    return false;
  }
  static boolean hasField(String cls, String name, Set<String> seen) throws Exception {
    if (cls == null || !seen.add(cls)) return false;
    ClassNode c = load(cls); if (c == null) return false;
    for (FieldNode f : c.fields) if (f.name.equals(name)) return true;
    if (hasField(c.superName, name, seen)) return true;
    for (String i : c.interfaces) if (hasField(i, name, seen)) return true;
    return false;
  }
  static FieldNode findField(String cls, String name, Set<String> seen) throws Exception {
    if (cls == null || !seen.add(cls)) return null;
    ClassNode c = load(cls); if (c == null) return null;
    for (FieldNode f : c.fields) if (f.name.equals(name)) return f;
    FieldNode r = findField(c.superName, name, seen); if (r != null) return r;
    for (String i : c.interfaces) { r = findField(i, name, seen); if (r != null) return r; }
    return null;
  }
  static Map<String,Object> ann(AnnotationNode a) {
    Map<String,Object> m = new LinkedHashMap<>();
    if (a.values != null) for (int i = 0; i < a.values.size(); i += 2) m.put((String) a.values.get(i), a.values.get(i + 1));
    return m;
  }
  @SuppressWarnings("unchecked")
  static List<Object> list(Object o) { if (o == null) return List.of(); return o instanceof List ? (List<Object>) o : List.of(o); }
  static int ok = 0, bad = 0, mixins = 0, injectors = 0;
  static void err(String s) { bad++; System.out.println("MISS " + s); }

  public static void main(String[] a) throws Exception {
    mc = new ZipFile(a[0]);
    Path classes = Paths.get(a[1]);
    for (int i = 4; i < a.length; i++) libs.add(new ZipFile(a[i]));
    List<String> names = Files.readAllLines(Paths.get(a[2])); // mixin class FQNs (slash form)
    String filter = a[3];
    for (String mn : names) {
      Path p = classes.resolve(mn + ".class");
      if (!Files.exists(p)) { err(mn + " class file missing"); continue; }
      ClassNode mx = new ClassNode(); new ClassReader(Files.readAllBytes(p)).accept(mx, ClassReader.SKIP_FRAMES);
      List<String> targets = new ArrayList<>();
      if (mx.invisibleAnnotations != null) for (AnnotationNode an : mx.invisibleAnnotations) if (an.desc.equals("Lorg/spongepowered/asm/mixin/Mixin;")) {
        Map<String,Object> m = ann(an);
        for (Object o : list(m.get("value"))) targets.add(((Type) o).getInternalName());
        for (Object o : list(m.get("targets"))) targets.add(((String) o).replace('.', '/'));
      }
      mixins++;
      List<ClassNode> tcs = new ArrayList<>();
      for (String t : targets) { ClassNode c = load(t); if (c == null) err(mn + " target class missing " + t); else tcs.add(c); }
      if (tcs.isEmpty()) continue;
      for (MethodNode mm : mx.methods) {
        List<AnnotationNode> all = new ArrayList<>();
        if (mm.invisibleAnnotations != null) all.addAll(mm.invisibleAnnotations);
        if (mm.visibleAnnotations != null) all.addAll(mm.visibleAnnotations);
        for (AnnotationNode an : all) {
          String d = an.desc; Map<String,Object> m = ann(an);
          String simple = d.substring(d.lastIndexOf('/') + 1, d.length() - 1);
          boolean inj = Set.of("Inject","Redirect","WrapMethod","WrapOperation","ModifyArg","ModifyArgs","ModifyVariable","ModifyConstant","WrapWithCondition","ModifyExpressionValue","ModifyReturnValue").contains(simple);
          if (inj) {
            injectors++;
            List<Object> ms = list(m.get("method"));
            for (Object mo : ms) {
              String ref = (String) mo; String mname = ref, mdesc = null;
              int pi = ref.indexOf('(');
              if (pi >= 0) { mname = ref.substring(0, pi); mdesc = ref.substring(pi); }
              if (mname.startsWith("L") && mname.contains(";")) mname = mname.substring(mname.indexOf(';') + 1);
              boolean found = false; List<MethodNode> hit = new ArrayList<>();
              for (ClassNode tc : tcs) for (MethodNode tm : tc.methods) if (tm.name.equals(mname) && (mdesc == null || tm.desc.equals(mdesc))) { found = true; hit.add(tm); }
              if (!found) { err(mn + "#" + mm.name + " @" + simple + " method '" + ref + "' not in " + targets); continue; }
              ok++;
              // At targets
              List<AnnotationNode> ats = new ArrayList<>();
              for (String key : List.of("at", "slice")) {}
              for (Object ao : list(m.get("at"))) ats.add((AnnotationNode) ao);
              for (AnnotationNode at : ats) {
                Map<String,Object> am = ann(at);
                String kind = (String) am.get("value"); String tg = (String) am.get("target");
                if (tg == null || kind == null) continue;
                if (!(kind.equals("INVOKE") || kind.equals("INVOKE_ASSIGN") || kind.equals("FIELD") || kind.equals("NEW"))) continue;
                String owner, nm, ds = null;
                if (tg.startsWith("L")) { owner = tg.substring(1, tg.indexOf(';')); String r = tg.substring(tg.indexOf(';') + 1); int q = r.indexOf('('); int c2 = r.indexOf(':'); if (kind.equals("FIELD") || q < 0) { nm = r.substring(0, c2 > 0 ? c2 : r.length()); } else { nm = r.substring(0, q); ds = r.substring(q); } }
                else { int dot = tg.indexOf('.'); if (dot < 0) continue; owner = tg.substring(0, dot); String r = tg.substring(dot + 1); int q = r.indexOf('('); if (q < 0) { nm = r; } else { nm = r.substring(0, q); ds = r.substring(q); } }
                if (kind.equals("NEW")) continue;
                boolean present = false;
                for (MethodNode tm : hit) for (AbstractInsnNode in : tm.instructions) {
                  if (in instanceof MethodInsnNode mi && mi.name.equals(nm) && (ds == null || mi.desc.equals(ds)) && (mi.owner.equals(owner) || true)) present = true;
                  if (in instanceof FieldInsnNode fi && fi.name.equals(nm)) present = true;
                  if (in instanceof InvokeDynamicInsnNode) { for (Object ba : ((InvokeDynamicInsnNode) in).bsmArgs) if (ba instanceof Handle h && h.getName().equals(nm)) present = true; }
                }
                if (!present) err(mn + "#" + mm.name + " @At " + kind + " target '" + tg + "' not invoked in " + ref); else ok++;
              }
            }
          } else if (simple.equals("Shadow")) {
            String nm = mm.name; Object pf = m.get("prefix"); String prefix = pf == null ? "shadow$" : (String) pf;
            if (nm.startsWith(prefix)) nm = nm.substring(prefix.length());
            boolean f = false; for (ClassNode tc : tcs) if (hasMethod(tc.name, nm, mm.desc, new HashSet<>())) f = true;
            if (!f) err(mn + " @Shadow method " + nm + mm.desc + " not in " + targets); else ok++;
          } else if (simple.equals("Overwrite")) {
            boolean f = false; for (ClassNode tc : tcs) if (hasMethod(tc.name, mm.name, mm.desc, new HashSet<>())) f = true;
            if (!f) err(mn + " @Overwrite " + mm.name + mm.desc + " not in " + targets); else ok++;
          } else if (simple.equals("Invoker") || simple.equals("Accessor")) {
            String v = (String) m.get("value"); String nm = v;
            boolean inv = simple.equals("Invoker");
            if (nm == null) { nm = mm.name; String[] pres = inv ? new String[]{"invoke","call"} : new String[]{"get","set","is"}; boolean fnd = false; for (String pre : pres) if (nm.startsWith(pre) && nm.length() > pre.length()) { nm = Character.toLowerCase(nm.charAt(pre.length())) + nm.substring(pre.length() + 1); fnd = true; break; } if (!fnd && !inv) { err(mn + " @Accessor " + mm.name + " has no value and no get/set/is prefix"); continue; } }
            Type[] ptypes = Type.getArgumentTypes(mm.desc); Type rt = Type.getReturnType(mm.desc);
            boolean ok1 = false; String why = "";
            for (ClassNode tc : tcs) {
              if (inv) {
                // match name + argument descriptor (skip coerced params: compare only non-coerced exactly)
                boolean[] coerce = new boolean[ptypes.length];
                if (mm.invisibleParameterAnnotations != null) for (int pi2 = 0; pi2 < mm.invisibleParameterAnnotations.length && pi2 < coerce.length; pi2++) if (mm.invisibleParameterAnnotations[pi2] != null) for (AnnotationNode pa : mm.invisibleParameterAnnotations[pi2]) if (pa.desc.endsWith("/Coerce;")) coerce[pi2] = true;
                boolean found = false;
                for (ClassNode cc = tc; cc != null && !found; ) {
                  for (MethodNode tm : cc.methods) if (tm.name.equals(nm)) {
                    Type[] tp = Type.getArgumentTypes(tm.desc); if (tp.length != ptypes.length) { why += " [argc " + tm.desc + "]"; continue; }
                    boolean allm = true; for (int k = 0; k < tp.length; k++) if (!coerce[k] && !tp[k].equals(ptypes[k])) allm = false;
                    if (!allm) { why += " [args " + tm.desc + "]"; continue; }
                    if (!nm.equals("<init>") && !Type.getReturnType(tm.desc).equals(rt) && !(rt.getSort() == Type.OBJECT && Type.getReturnType(tm.desc).getSort() == Type.OBJECT && rt.getInternalName().equals("java/lang/Object") && false)) { why += " [ret " + tm.desc + "]"; if (!rt.getDescriptor().equals("V") || true) { if (!Type.getReturnType(tm.desc).equals(rt)) continue; } }
                    found = true; break;
                  }
                  cc = cc.superName == null ? null : load(cc.superName);
                }
                if (!found) for (ClassNode tc2 : List.of(tc)) { Set<String> sn = new HashSet<>(); if (hasMethod(tc2.name, nm, null, sn)) { /* inherited via interface: accept name only */ } }
                ok1 |= found;
              } else {
                FieldNode fnode = findField(tc.name, nm, new HashSet<>());
                if (fnode == null) { why += " [no field]"; continue; }
                Type ft = Type.getType(fnode.desc);
                boolean getter = ptypes.length == 0 && rt.getSort() != Type.VOID;
                boolean setter = ptypes.length == 1 && rt.getSort() == Type.VOID;
                if (!getter && !setter) { why += " [bad accessor signature]"; continue; }
                boolean coerced = false;
                if (setter && mm.invisibleParameterAnnotations != null && mm.invisibleParameterAnnotations.length > 0 && mm.invisibleParameterAnnotations[0] != null) for (AnnotationNode pa : mm.invisibleParameterAnnotations[0]) if (pa.desc.endsWith("/Coerce;")) coerced = true;
                Type got = getter ? rt : ptypes[0];
                if (!coerced && !got.equals(ft)) { why += " [field is " + fnode.desc + " accessor uses " + got.getDescriptor() + "]"; continue; }
                ok1 = true;
              }
            }
            if (!ok1) err(mn + " @" + simple + " " + mm.name + mm.desc + " -> " + nm + " not matching in " + targets + why); else ok++;
          }
        }
      }
      for (FieldNode fn : mx.fields) {
        List<AnnotationNode> all = new ArrayList<>(); if (fn.invisibleAnnotations != null) all.addAll(fn.invisibleAnnotations);
        for (AnnotationNode an : all) if (an.desc.endsWith("/Shadow;")) {
          boolean f = false; String why = ""; for (ClassNode tc : tcs) { FieldNode tf = findField(tc.name, fn.name, new HashSet<>()); if (tf == null) continue; if (tf.desc.equals(fn.desc)) f = true; else why += " [field is " + tf.desc + " shadow uses " + fn.desc + "]"; }
          if (!f) err(mn + " @Shadow field " + fn.name + " not matching in " + targets + why); else ok++;
        }
      }
    }
    System.out.println("mixins=" + mixins + " injectors=" + injectors + " checks_ok=" + ok + " MISSES=" + bad);
  }
}
